# Computer lab logo animation slots (src/main/computer_lab.c): g_ClLogoCur = the slot every computer_logo_* step
# works on (+1 behaviour, +2 sub-step, +8/+C position, +0x2C angle per config/globals_main_2.txt).
struct ClLogoSlot
view CL_LOGO = ((ClLogoSlot *)g_ClLogoCur)
base g_ClLogoCur
define
0x00 s8  active
0x01 u8  behaviour     # logo behaviour (step table selector)
0x02 u8  substep       # computer_logo_hub_step_tbl index
0x08 s32 x             # 16.16 position
0x0C s32 y
0x14 u16 timer         # set to 3, counted down
0x16 s16 phase_signal         # 0 / 1
0x18 void * parent
0x1C s8  tex_u         # sprite: texture u (0x20)
0x1D s8  tex_v         # texture v (0x18)
0x1E s16 width         # 0x20
0x20 s16 height        # 0x28
0x22 s16 ot_pri         # 5
0x24 s16 pivot_x         # 0x10
0x26 s16 pivot_y         # 0x28
0x28 s16 scale_x       # 0x1000 = 1.0
0x2A s16 scale_y
0x2C s32 angle
