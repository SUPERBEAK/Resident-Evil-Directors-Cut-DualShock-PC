# Ending script records (src/ending/ending.c), PS1-only (no PC counterpart).
# g_EndingPicturePtr: current 12-byte record of ending_picture_seq_tbl (ending_picture_sequence_update);
# a record with show_frame 0xFFFF ends the script.
struct EndingPictureCue
view END_CUE = ((EndingPictureCue *)g_EndingPicturePtr)
base g_EndingPicturePtr
define
0x00 u16 show_frame    # g_EndingStrFrame reached: place the curtains (state 1 -> 2)
0x02 u16 move_frame    # g_EndingStrFrame reached: start moving them, next record (state 2)
0x04 s16 box_x         # g_EndingBoxB_x = box_x - 0xA0
0x06 s16 box_y         # g_EndingBoxA_y = box_y - 0x78
0x08 u16 box_vel_x     # -> picture slot 0x14 (added to g_EndingBoxB_x per frame)
0x0A u16 box_vel_y     # -> picture slot 0x16 (added to g_EndingBoxA_y per frame)
