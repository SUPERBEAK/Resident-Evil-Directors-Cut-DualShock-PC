# Room collision boundary: one 12-byte record of the RDT collision block (RDT +0x4C: a list of block
# pointers, each to records of this kind; PC RDT_Boundary). SCD 0x30 boundary_set rewrites one.
# The max corner comes first (PC decomp); the coordinates are unsigned.
struct RoomBoundary
macro BOUNDARY
param scd_op_30_boundary_set obj
define
0x00 u16 x_max
0x02 u16 z_max
0x04 u16 x_min
0x06 u16 z_min
0x08 u16 type     # low byte: shape handler index; bits 8-14 floor/step magnitude, bit 15 step down
0x0A u16 flags    # low byte: floor/step fine value; 0x100 blocks movement; 0x0F00 set by SCD 0x30
noaddr
