# SCD 0x1B em_set record (22 bytes; RDT page script_em_set_t, PC cmd_enemy_set): an enemy slot.
struct ScdEmSet
noaddr
macro SCD_EM_SET
param scd_op_1b_enemy_set p
param scd_op_1b_enemy_set p2
param scd_op_1b_enemy_set p3
define
0x01 u8 id            # -> entity id (model type)
0x02 u8 behavior      # -> behavior_flags
0x03 u8 killed        # enemy flag bit (0x800C86BC; set = dead, skip) and death_event_id; 0xFF = none
0x04 u8 no_restore         # 0: restore_saved_enemy_state first; else entity +0x161 |= 0x80 and always init
0x05 u8 sca_count     # SCA pool entries reserved (g_ScaHitPoolPtr += n * 6)
0x06 u16 rot_x        # -> rot_x
0x08 u16 angle        # -> angle
0x0A u16 angle_z      # -> angle_z
0x0C u16 x            # -> pos_x
0x0E s16 y            # -> pos_y (0x800C52DC + slot * 0x18C)
0x10 u16 z            # -> pos_z
0x12 u8 slot          # low nibble: enemy slot (0x800C52A4 + n * 0x18C)
0x13 u8 anim_id       # -> animation_id
0x14 u8 anim_frame    # -> animation_frame_id
0x15 u8 sound_set         # high nibble of entity +0x161
param scd_op_1b_enemy_set LIT(void *, g_ScdIp)
param scd_op_1b_enemy_set LIT(s32, g_ScdIp)
